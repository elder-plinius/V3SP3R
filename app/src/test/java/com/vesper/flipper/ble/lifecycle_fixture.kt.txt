open class Context {
    val permissions = mutableSetOf("connect", "scan", "bluetooth", "admin")
    val usb = UsbManager()
    val bluetooth = BluetoothManager()
    val notifications = android.app.NotificationManager()
    var dispatches = 0
    var dispatchFailure: RuntimeException? = null
    var binds = 0
    var unbinds = 0
    var bindResult = true
    var bindFailure: RuntimeException? = null
    val connections = mutableListOf<ServiceConnection>()
    val autoCreateBindings = mutableSetOf<ServiceConnection>()
    var startRequested = false
    var fgRequired = false
    var fgWaiting = false
    var foregroundWatchdogScheduled = false
    var foregroundStartFailure = false
    var destroying = false

    fun runForegroundWatchdog() {
        if (foregroundWatchdogScheduled && fgRequired && fgWaiting && !destroying) {
            fgWaiting = false
            foregroundStartFailure = true
        }
    }
    fun getSystemService(name: String): Any? = when(name) {
        USB_SERVICE -> usb
        BLUETOOTH_SERVICE -> bluetooth
        NOTIFICATION_SERVICE -> notifications
        else -> null
    }
    fun startForegroundService(intent: Intent): ComponentName {
        val component = startService(intent)
        // Model an accepted request delivered to the app: Android arms the
        // foreground-start timeout before onStartCommand executes.
        fgRequired = true
        fgWaiting = true
        foregroundWatchdogScheduled = true
        return component
    }
    fun startService(intent: Intent): ComponentName {
        dispatches++
        dispatchFailure?.let { throw it }
        startRequested = true
        return ComponentName()
    }
    fun bindService(intent: Intent, connection: ServiceConnection, flags: Int): Boolean {
        binds++
        connections.add(connection)
        bindFailure?.let { throw it }
        if (bindResult && flags and BIND_AUTO_CREATE != 0) autoCreateBindings.add(connection)
        return bindResult
    }
    fun unbindService(connection: ServiceConnection) {
        unbinds++
        autoCreateBindings.remove(connection)
    }
    companion object {
        const val USB_SERVICE = "usb"
        const val BLUETOOTH_SERVICE = "bluetooth"
        const val NOTIFICATION_SERVICE = "notification"
        const val BIND_AUTO_CREATE = 1
    }
}

class FlipperBleService : Service() {
    val packageName = "test"
    var pendingUsbPermissionRequest: CompletableDeferred<Boolean>? = null
    val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    var scanStopFailure: SecurityException? = null
    var scanStops = 0
    var disconnects = 0
    var disconnectedWhileForeground = false
    fun unregisterUsbReceivers() {}
    fun usbPermissionPendingIntentFlags() = 0
    suspend fun ensureUsbForTest(device: UsbDevice) = ensureUsbPermission(usb, device)
    var revokeDuringPromotion = false
    var promotionFailure: RuntimeException? = null
    var promotions = 0
    var successfulPromotions = 0
    var stops = 0
    var removals = 0
    var foregroundActive = false
    var bluetoothAdapter: BluetoothAdapter? = null
    var bleScanner: BluetoothLeScanner? = null
    val connectionState = MutableStateFlow<ConnectionState>(ConnectionState.Disconnected)
    val discoveredDevices = MutableStateFlow<List<FlipperDevice>>(emptyList())
    val _connectionState = connectionState
    val _discoveredDevices = discoveredDevices
    var broadScanStarted = false
    val scanCallback = Any()
    fun seedBondedFlippersIntoDiscoveredList() {}
    fun startBroadScan() {}
    fun notifyForTest() = updateNotification()
    val connectedDevice = MutableStateFlow<FlipperDevice?>(null)
    val cliCapabilityStatus = MutableStateFlow(CliCapabilityStatus())
    inner class LocalBinder : IBinder { fun getService() = this@FlipperBleService }
    fun createNotification() = Unit
    fun startForeground(id: Int, notification: Unit) {
        promotions++
        // Android 15 ActiveServices.setServiceForegroundInnerLocked clears the
        // accepted-start obligation BEFORE connectedDevice permission validation
        // and background-start rejection. stopForeground does not do this.
        fgRequired = false
        fgWaiting = false
        foregroundWatchdogScheduled = false
        if (revokeDuringPromotion) permissions.clear()
        promotionFailure?.let { throw it }
        if (Build.VERSION.SDK_INT >= 34 &&
            "connect" !in permissions && "scan" !in permissions &&
            usb.deviceList.values.none { usb.hasPermission(it) }
        ) throw SecurityException("connectedDevice transport permission missing")
        successfulPromotions++
    }
    fun stopSelf() {
        stops++
        startRequested = false
        // stopServiceTokenLocked cannot bring down a BIND_AUTO_CREATE-retained
        // service. In particular it cannot cancel its foreground-start timer.
        if (autoCreateBindings.isEmpty()) {
            destroying = true
            // bringDownServiceLocked reports an unmet obligation even on teardown.
            if (fgRequired) foregroundStartFailure = true
            fgRequired = false
            fgWaiting = false
            foregroundWatchdogScheduled = false
        }
    }
    fun stopSelf(startId: Int) { stopSelf() }
    fun stopForeground(flags: Int) { removals++ }

    fun stopScan() { scanStops++; scanStopFailure?.let { throw it } }
    fun connect(device: FlipperDevice) {}
    fun connectUsbIfAvailable() {}
    fun disconnect() { disconnects++; disconnectedWhileForeground = foregroundActive }
    suspend fun sendRawData(data: ByteArray) = Result.success(data)
    suspend fun probeCliCapability(force: Boolean) = CliCapabilityStatus()
    fun initializeForTest() = initializeBluetooth()
    // SERVICE_METHODS
    companion object {
        const val ACTION_START_FOREGROUND = "start"
        const val ACTION_STOP = "stop"
        const val NOTIFICATION_ID = 1
        const val STOP_FOREGROUND_REMOVE = 1
        const val TAG = "test"
        const val ACTION_USB_PERMISSION = "usb-permission"
        const val USB_PERMISSION_TIMEOUT_MS = 100L
        const val BROAD_SCAN_FALLBACK_DELAY_MS = 60000L
        const val SCAN_TIMEOUT_MS = 60000L
        val FLIPPER_SCAN_SERVICE_UUIDS = emptyList<java.util.UUID>()
        // COMPANION_METHODS
    }
}

class ExtractedServiceStartupTest {
    @Test fun acceptedUsbStartLosesEligibilityWhileBindingRetainsService() {
        val service = FlipperBleService().apply {
            permissions.clear()
            usb.deviceList["flipper"] = UsbDevice(true)
        }
        val manager = BleServiceManager(service)
        service.connections.single().onServiceConnected(null, service.LocalBinder())
        try {
            assertEquals(1, service.dispatches)
            assertTrue(service.startRequested)
            assertTrue(service.fgRequired)
            assertTrue(service.fgWaiting)
            assertTrue(service.foregroundWatchdogScheduled)
            assertEquals(1, service.autoCreateBindings.size)

            // The accepted request arrives after the only transport grant is lost.
            service.usb.deviceList.clear()
            assertEquals(Service.START_NOT_STICKY, service.onStartCommand(Intent("start"), 0, 1))
            assertEquals(1, service.stops)
            assertEquals(1, service.removals)
            assertFalse(service.startRequested)
            assertFalse(service.destroying)
            assertEquals(1, service.autoCreateBindings.size)
            assertSame(service, manager.getService())
            service.runForegroundWatchdog()
            assertFalse("Accepted foreground start left an undischarged watchdog while bound",
                service.foregroundStartFailure)
            assertFalse(service.fgRequired)
            assertFalse(service.fgWaiting)
            assertFalse(service.foregroundWatchdogScheduled)
            assertEquals(1, service.promotions)
            assertEquals(0, service.successfulPromotions)
            assertFalse(service.foregroundActive)
        } finally {
            manager.unbindService()
        }
    }
    @Test fun fakeBoundStopCannotDischargeAcceptedForegroundWatchdog() {
        val service = FlipperBleService()
        val manager = BleServiceManager(service)
        try {
            service.stopForeground(FlipperBleService.STOP_FOREGROUND_REMOVE)
            service.stopSelf(1)
            assertFalse(service.startRequested)
            assertFalse(service.destroying)
            assertTrue(service.fgRequired)
            assertTrue(service.fgWaiting)
            assertTrue(service.foregroundWatchdogScheduled)
            service.runForegroundWatchdog()
            assertTrue(service.foregroundStartFailure)
        } finally {
            manager.unbindService()
        }
    }
    @Test fun grantRetryThroughActualStartScanAcquiresScanner() {
        val service = FlipperBleService().apply { permissions.clear() }
        service.initializeForTest()
        assertNull(service.bleScanner)
        service.permissions.addAll(listOf("connect", "scan"))
        service.startScan()
        assertEquals(1, service.bluetooth.adapter.scanner.starts)
        service.onDestroy()
    }
    @Test fun boundOnlyOrDestroyedServiceDoesNotPostOrphanNotification() {

    @Test fun backgroundDispatchRestrictionIsContainedWithoutUnbinding() {

        val manager = BleServiceManager(context)

    }
    @Test fun backgroundPromotionRestrictionStopsNonSticky() {

class ExtractedServiceBindingTest {
    @Test fun deniedDeviceEntryStillBindsForUsbPermissionRequest() {
        val context = Context().apply { permissions.clear() }
        val manager = BleServiceManager(context)
        assertEquals(0, context.dispatches)
        assertEquals(1, context.binds)
        manager.unbindService()
    }
    @Test fun failedBindCanRetry() {
        val context = Context().apply { bindResult = false }
        val manager = BleServiceManager(context)
        context.bindResult = true
        manager.bindService()
        assertEquals(2, context.binds)
        assertEquals(1, context.unbinds)
        manager.unbindService()
    }
    @Test fun unbindBeforeConnectionReleasesPendingBinding() {
        val context = Context()
        val manager = BleServiceManager(context)
        manager.unbindService()
        assertEquals(1, context.unbinds)
        manager.bindService()
        assertEquals(2, context.binds)
        manager.unbindService()
    }
    @Test fun duplicateBindDoesNotRegisterTwice() {
        val context = Context()
        val manager = BleServiceManager(context)
        manager.bindService()
        assertEquals(1, context.binds)
        manager.unbindService()
    }
    @Test fun temporaryDisconnectRetainsRegistrationForAndroidReconnect() {
        val context = Context()
        val manager = BleServiceManager(context)
        val connection = context.connections.single()
        connection.onServiceDisconnected(null)
        manager.bindService()
        assertEquals(1, context.binds)
        assertEquals(0, context.unbinds)
        connection.onServiceConnected(null, FlipperBleService().LocalBinder())
        assertTrue(manager.isServiceBound.value)
        manager.unbindService()
    }
    @Test fun nullBindingIsNotReportedBoundAndCanRetry() {
        val context = Context()
        val manager = BleServiceManager(context)
        context.connections.single().onNullBinding(null)
        assertFalse(manager.isServiceBound.value)
        assertNull(manager.getService())
        manager.bindService()
        assertEquals(2, context.binds)
        assertEquals(1, context.unbinds)
        manager.unbindService()
    }
    @Test fun invalidBinderIsNotReportedBound() {
        val context = Context()
        val manager = BleServiceManager(context)
        context.connections.single().onServiceConnected(null, null)
        assertFalse(manager.isServiceBound.value)
        assertNull(manager.getService())
        manager.unbindService()
    }
    @Test fun bindingDeathReleasesRegistrationAndCanRetry() {
        val context = Context()
        val manager = BleServiceManager(context)
        context.connections.single().onBindingDied(null)
        manager.bindService()
        assertEquals(2, context.binds)
        assertEquals(1, context.unbinds)
        manager.unbindService()
    }
    @Test fun lateCallbackFromReleasedBindingCannotResurrectService() {
        val context = Context()
        val manager = BleServiceManager(context)
        val old = context.connections.single()
        manager.unbindService()
        manager.bindService()
        old.onServiceConnected(null, FlipperBleService().LocalBinder())
        assertFalse(manager.isServiceBound.value)
        assertNull(manager.getService())
        manager.unbindService()
    }
    @Test fun grantRetryPromotesWithoutUnbindingExistingTransport() {
        val context = Context().apply { permissions.clear() }
        val manager = BleServiceManager(context)
        val service = FlipperBleService()
        context.connections.single().onServiceConnected(null, service.LocalBinder())
        context.permissions.add("connect")
        manager.bindService()
        assertEquals(1, context.dispatches)
        assertEquals(1, context.binds)
        assertEquals(0, context.unbinds)
        assertSame(service, manager.getService())
        manager.unbindService()
    }
    @Test fun bindSecurityExceptionDoesNotPoisonRetry() {
        val context = Context().apply { bindFailure = SecurityException("bind denied") }
        val manager = BleServiceManager(context)
        context.bindFailure = null
        manager.bindService()
        assertEquals(2, context.binds)
        manager.unbindService()
    }
